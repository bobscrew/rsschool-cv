Code Example

Room Number Sorter

Problem

Room numbers like 1.1, 1.2, 1.21 sort incorrectly with default string comparison: it produces 1.1, 1.21, 1.2 because characters are compared one by one, not numeric segments. For engineering documentation this order is wrong and breaks cross-referencing.

Solution

A custom comparer that splits each number by ., compares segments as integers, and treats missing segments as zero. Implements both IComparer<string> and IEqualityComparer<string> so the same logic powers both Distinct() and OrderBy(). 


          using System;
          using System.Collections.Generic;
          using System.Linq;
          class Program
          {
            static void Main()
            {
              List<string> values = new List<string> { "1", "3", "3", "2.2", "3", "1.1" };
              var distinctValues = values.Distinct(new DotSeparatedComparer()).ToList();
              var sortedValues = distinctValues.OrderBy(v => v, new DotSeparatedComparer()).ToList();
              foreach (var value in sortedValues)
              {
                Console.WriteLine(value);
              }
            }
          }
          
          class DotSeparatedComparer : IComparer<string>, IEqualityComparer<string>
          {
            public int Compare(string x, string y)
            {
              var xParts = x.Split('.').Select(int.Parse).ToArray();
              var yParts = y.Split('.').Select(int.Parse).ToArray();
              int length = Math.Max(xParts.Length, yParts.Length);
              for (int i = 0; i < length; i++)
              {
                int xPart = i < xParts.Length ? xParts[i] : 0;
                int yPart = i < yParts.Length ? yParts[i] : 0;
                int comparison = xPart.CompareTo(yPart);
                if (comparison != 0)
                {
                  return comparison;
                }
              }
              return 0;
            }

            public bool Equals(string x, string y)
            {
              return Compare(x, y) == 0;
            }

            public int GetHashCode(string obj)
            {
              var parts = obj.Split('.').Select(int.Parse).ToArray();
              int hash = 17;
              foreach (var part in parts)
              {
                hash = hash * 31 + part.GetHashCode();
              }
              return hash;
            }
          }
